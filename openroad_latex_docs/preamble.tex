% ------------------------------------------------------------
% Convenience macros (use these everywhere for consistency)
% ------------------------------------------------------------
% File path in the OpenROAD tree, e.g. \file{src/odb/include/odb/db.h}
\newcommand{\file}[1]{\texttt{\textcolor{orblue}{#1}}}
% A Tcl/shell command, e.g. \cmd{global_placement}
\newcommand{\cmd}[1]{\texttt{\textbf{#1}}}
% A C++ identifier (class / function / field), e.g. \id{dbInst}
\newcommand{\id}[1]{\texttt{#1}}
% A module / package directory, e.g. \pkg{src/gpl}
\newcommand{\pkg}[1]{\texttt{\textcolor{orteal}{#1}}}

% Boxed "at a glance" note
\newcommand{\glance}[1]{%
  \begin{center}
  \fcolorbox{orrule}{orlight}{%
    \parbox{0.95\linewidth}{\small\textbf{At a glance.} #1}%
  }
  \end{center}%
}
