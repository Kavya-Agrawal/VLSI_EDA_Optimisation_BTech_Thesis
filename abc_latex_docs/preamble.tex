% ------------------------------------------------------------
% Convenience macros (use these everywhere for consistency)

% File path in the ABC tree, e.g. \file{src/base/abc/abc.h}
\newcommand{\file}[1]{\texttt{\textcolor{abcblue}{#1}}}
% A shell / REPL command, e.g. \cmd{resyn2}
\newcommand{\cmd}[1]{\texttt{\textbf{#1}}}
