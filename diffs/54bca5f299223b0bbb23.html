113// Speculative clustering can restore connectivity without copying the forest.
114// Path compression is intentionally omitted so each union has a bounded log.
115class RollbackSet
116{
117 public:
118  explicit RollbackSet(std::size_t size) : parent_(size), sizes_(size, 1)
119  {
120    std::iota(parent_.begin(), parent_.end(), 0);
121  }
122 
123  std::size_t find(std::size_t node) const
124  {
125    if (node >= parent_.size()) {
126      throw std::out_of_range("RollbackSet node");
127    }
128    while (parent_[node] != node) {
129      node = parent_[node];
130    }
131    return node;
132  }
133 
134  std::size_t checkpoint() const { return history_.size(); }
135 
136  bool unite(std::size_t a, std::size_t b)
137  {
138    a = find(a);
139    b = find(b);
140    if (a == b) {
141      return false;
142    }
143    if (sizes_[a] < sizes_[b]) {
144      std::swap(a, b);
145    }
146    history_.push_back({a, b, sizes_[a]});
147    parent_[b] = a;
148    sizes_[a] += sizes_[b];
149    return true;
150  }

152  void rollback(std::size_t checkpoint)
153  {
154    if (checkpoint > history_.size()) {
155      throw std::out_of_range("RollbackSet future checkpoint");
156    }

158      const auto record = history_.back();
159      history_.pop_back();

162    }
163  }
