103inline std::uint64_t mortonKey(const Point& p, const Bounds& bounds)
104{
105  const auto quantize = [](double value, double low, double high) {
106    if (high <= low) {

108    }

113         | (spreadBits(quantize(p.y, bounds.ymin, bounds.ymax)) << 1);

116inline std::vector<Point> convexHull(std::vector<Point> points)

368// Double-tree preorder gives a <=2 metric tour, prior to optional local search.
369inline std::vector<std::size_t> preorderTour(std::size_t vertices,
370                                            const std::vector<SpatialEdge>& tree)
371{
372  if (!vertices) {
373    return {};
374  }
375  std::vector<std::vector<std::size_t>> adjacent(vertices);
376  for (const auto& edge : tree) {
377    adjacent.at(edge.from).push_back(edge.to);
378    adjacent.at(edge.to).push_back(edge.from);
379  }
380  std::vector<bool> seen(vertices, false);
381  std::vector<std::size_t> order;
382  std::vector<std::size_t> pending{0};
383  while (!pending.empty()) {
384    const auto node = pending.back();
385    pending.pop_back();
386    if (seen[node]) {
387      continue;
388    }
389    seen[node] = true;
390    order.push_back(node);
391    auto& next = adjacent[node];
392    std::sort(next.rbegin(), next.rend());
393    pending.insert(pending.end(), next.begin(), next.end());
394  }
395  if (order.size() != vertices) {
396    throw std::invalid_argument("Tour needs a connected tree");
397  }
398  return order;
399}

422                                      std::vector<std::size_t> tour,
423                                      unsigned max_passes)
424{

428      for (std::size_t j = i + 2; j < tour.size(); ++j) {

432        const double after = manhattan(points.at(tour[i]), points.at(tour[j]))

447inline std::vector<std::size_t> farthestFirst(const std::vector<Point>& points,
448                                              std::size_t count)
449{
450  if (points.empty() || count == 0) {
451    return {};
452  }
453  count = std::min(count, points.size());
454  std::vector<std::size_t> centers{0};
455  std::vector<bool> chosen(points.size(), false);
456  chosen[0] = true;
457  std::vector<double> distances(points.size(), std::numeric_limits<double>::infinity());
458  while (centers.size() < count) {
459    for (std::size_t i = 0; i < points.size(); ++i) {

469    }

483inline std::vector<std::size_t> spatialOrder(const std::vector<Point>& points,
484                                            PolicyConfig config = policyConfig())
485{
486  std::vector<std::size_t> order(points.size());
487  std::iota(order.begin(), order.end(), 0);
488  if (points.size() < 2) {
489    return order;
490  }
491  Bounds bounds;
492  for (const auto& point : points) {
493    if (!std::isfinite(point.x) || !std::isfinite(point.y)) {
494      throw std::invalid_argument("Spatial ordering needs finite coordinates");
495    }
496    bounds.extend(point);
497  }
498  if (config.spatial_mode == 0) {
499    return order;
500  }
501  if (config.spatial_mode == 2 && points.size() <= 512) {
502    // Dense MST is deliberately capped to bound quadratic memory.
503    return twoOpt(points, preorderTour(points.size(), denseManhattanTree(points)), 2);
504  }
505  const KdTree index(points);
506  std::vector<std::uint64_t> keys;
507  std::vector<double> isolation;
508  for (const auto& point : points) {
509    keys.push_back(mortonKey(point, bounds));
510    isolation.push_back(index.nearest(point, point.id).first);
511  }
512  std::stable_sort(order.begin(), order.end(), [&](auto a, auto b) {
513    if (keys[a] != keys[b]) {
514      return keys[a] < keys[b];
515    }
516    if (isolation[a] != isolation[b]) {
517      return isolation[a] < isolation[b];
518    }
519    return a < b;
520  });
521  return order;
522}
