61// Zero-based public indices; half-open prefix/range conventions.
62class Fenwick
63{
64 public:
65  explicit Fenwick(std::size_t size) : tree_(size + 1, 0.0) {}
66  std::size_t size() const { return tree_.size() - 1; }
67 
68  void add(std::size_t index, double delta)
69  {
70    if (index >= size()) {
71      throw std::out_of_range("Fenwick point index");
72    }
73    for (++index; index < tree_.size(); index += index & -index) {
74      tree_[index] += delta;
75    }
76  }
77 
78  double prefix(std::size_t end) const
79  {
80    if (end > size()) {
81      throw std::out_of_range("Fenwick prefix end");
82    }
83    double value = 0;
84    for (; end > 0; end -= end & -end) {
85      value += tree_[end];
86    }
87    return value;
88  }
89 
90  double range(std::size_t begin, std::size_t end) const
91  {
92    if (begin > end) {
93      throw std::invalid_argument("Fenwick reversed range");
94    }
95    return prefix(end) - prefix(begin);
96  }
97 
98  // First zero-based index whose inclusive prefix reaches target.
99  // All point values must be nonnegative for this selection operation.
100  std::size_t select(double target) const
101  {
102    if (target <= 0) {
103      return 0;
104    }
105    std::size_t index = 0;
106    std::size_t step = 1;
107    while (step <= size() / 2) {
108      step *= 2;
109    }
110    for (; step > 0; step /= 2) {
111      const auto next = index + step;
112      if (next <= size() && tree_[next] < target) {
113        target -= tree_[next];
114        index = next;
115      }
116    }
117    return index;  // size() is the sentinel when target exceeds total.
118  }
119 
120 private:
121  std::vector<double> tree_;
122};

148// Associative aggregate supports WNS and TNS from the same indexed tree.
149class SlackTree
150{
151 public:
152  explicit SlackTree(const std::vector<double>& slacks) : count_(slacks.size())
153  {
154    while (size_ < count_) {
155      size_ *= 2;
156    }
157    tree_.resize(2 * size_);
158    for (std::size_t i = 0; i < count_; ++i) {
159      tree_[size_ + i] = leaf(slacks[i]);
160    }
161    for (auto i = size_; i-- > 1;) {
162      tree_[i] = merge(tree_[2 * i], tree_[2 * i + 1]);
163    }
164  }
165 
166  void update(std::size_t index, double slack)
167  {
168    if (index >= count_) {
169      throw std::out_of_range("SlackTree point index");
170    }
171    index += size_;
172    tree_[index] = leaf(slack);
173    while (index > 1) {
174      index /= 2;
175      tree_[index] = merge(tree_[2 * index], tree_[2 * index + 1]);
176    }
177  }
178 
179  SlackSummary query(std::size_t begin, std::size_t end) const
180  {
181    if (begin > end || end > count_) {
182      throw std::out_of_range("SlackTree query range");
183    }
184    SlackSummary left, right;
185    for (begin += size_, end += size_; begin < end; begin /= 2, end /= 2) {
186      if (begin & 1) {
187        left = merge(left, tree_[begin++]);
188      }
189      if (end & 1) {
190        right = merge(tree_[--end], right);
191      }
192    }
193    return merge(left, right);
194  }
195 
196 private:
197  static SlackSummary leaf(double slack)
198  {
199    return {slack, std::min(0.0, slack), 1};
200  }
201  std::size_t count_;
202  std::size_t size_ = 1;
203  std::vector<SlackSummary> tree_;
204};

231inline BudgetSelection budgetDp(const std::vector<BudgetItem>& items, unsigned budget)
232{
233  struct Link
234  {
235    std::size_t id;
236    std::shared_ptr<const Link> previous;
237  };
238  struct State
239  {
240    double benefit = 0;
241    unsigned cost = 0;
242    std::shared_ptr<const Link> tail;
243  };
244  std::vector<State> states(budget + 1);
245  for (const auto& item : items) {
246    if (!item.cost || !std::isfinite(item.benefit)) {
247      throw std::invalid_argument("DP needs positive cost and finite benefit");
248    }
249    if (item.cost > budget || item.benefit <= 0) {
250      continue;
251    }
252    for (unsigned capacity = budget; capacity >= item.cost; --capacity) {
253      const State previous = states[capacity - item.cost];
254      const double proposal = previous.benefit + item.benefit;
255      if (proposal > states[capacity].benefit) {
256        states[capacity] = {proposal, previous.cost + item.cost,
257                           std::make_shared<Link>(Link{item.id, previous.tail})};
258      }
259    }
260  }
261  BudgetSelection result{states[budget].benefit, states[budget].cost, {}};
262  for (auto link = states[budget].tail; link; link = link->previous) {
263    result.ids.push_back(link->id);
264  }
265  std::reverse(result.ids.begin(), result.ids.end());
266  return result;
267}

269inline BudgetSelection budgetGreedy(std::vector<BudgetItem> items, unsigned budget)
270{
271  for (const auto& item : items) {
272    if (!item.cost || !std::isfinite(item.benefit)) {
273      throw std::invalid_argument("Greedy needs positive cost and finite benefit");
274    }
275  }
276  std::stable_sort(items.begin(), items.end(), [](const auto& a, const auto& b) {
277    return a.benefit / a.cost > b.benefit / b.cost;
278  });
279  BudgetSelection result;
280  for (const auto& item : items) {
281    if (item.benefit > 0 && item.cost <= budget - result.cost) {
282      result.ids.push_back(item.id);
283      result.cost += item.cost;
284      result.benefit += item.benefit;
285    }
286  }
287  return result;

310inline std::vector<RepairFeature> extractRepairFeatures(
311    const std::vector<double>& loads, const PolicyConfig& config)
312{
313  double scale = 0;
314  for (double load : loads) {
315    if (!std::isfinite(load)) {
316      throw std::invalid_argument("Nonfinite path-driver load");
317    }
318    scale = std::max(scale, std::abs(load));
319  }
320  scale = std::max(scale, 1e-30);
321  Fenwick prefix(loads.size());
322  std::vector<double> negative_load;
323  for (std::size_t i = 0; i < loads.size(); ++i) {
324    const double normalized = std::max(0.0, loads[i] / scale);
325    prefix.add(i, normalized);
326    negative_load.push_back(-normalized);
327  }
328  SlackTree tree(negative_load);
329  std::vector<RepairFeature> result;
330  for (std::size_t i = 0; i < loads.size(); ++i) {
331    const auto left = i > config.neighborhood ? i - config.neighborhood : 0;
332    const auto right = std::min(loads.size(), i + config.neighborhood + 1);
333    const double normalized = -negative_load[i];
334    const double accumulated = prefix.prefix(i + 1) / (i + 1);
335    const double peak = -tree.query(left, right).worst;
336    const double position = loads.size() > 1 ? double(i) / (loads.size() - 1) : 0;
337    const double density = prefix.range(left, right) / (right - left);
338    const double score = config.load_weight * normalized
339                         + config.prefix_weight * accumulated
340                         + config.neighborhood_weight * peak
341                         + config.density_weight * density
342                         + config.late_weight * position
343                         + config.critical_weight * normalized * normalized;
344    result.push_back({i, normalized, accumulated, peak, score});
345  }
346  return result;
347}

349inline std::vector<std::size_t> repairOrder(const std::vector<double>& loads,
350                                           PolicyConfig config = policyConfig())
351{
352  const auto features = extractRepairFeatures(loads, config);
353  std::vector<BudgetItem> items;
354  for (const auto& feature : features) {
355    // Cost is a bounded illustrative repair-effort estimate, not cell area.
356    const auto cost = 1u + static_cast<unsigned>(3 * feature.normalized_load);
357    items.push_back({feature.original, cost, feature.score});
358  }
359  const auto selected = config.use_dp ? budgetDp(items, config.budget)
360                                      : budgetGreedy(items, config.budget);
361  std::vector<bool> preferred(loads.size(), false);
362  for (auto id : selected.ids) {
363    preferred[id] = true;
364  }
365  std::vector<std::size_t> order(loads.size());
366  std::iota(order.begin(), order.end(), 0);
367  std::stable_sort(order.begin(), order.end(), [&](auto a, auto b) {
368    if (preferred[a] != preferred[b]) {
369      return bool(preferred[a]);
370    }
371    if (features[a].score != features[b].score) {
372      return features[a].score > features[b].score;
373    }
374    return a < b;
375  });
376  return order;  // Every original target remains exactly once.
377}
