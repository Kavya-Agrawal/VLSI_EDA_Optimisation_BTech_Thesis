40def percentile(values, percentage):
41    if not values or not 0 <= percentage <= 100:
42        raise ValueError('Percentile needs data and a percentage in [0, 100]')
43    ordered = sorted(values)
44    position = (len(ordered) - 1) * percentage / 100
45    low, high = math.floor(position), math.ceil(position)
46    return ordered[low] + (ordered[high] - ordered[low]) * (position - low)

49def describe(values, field, unit):
50    if not values or not all(math.isfinite(value) for value in values):
51        raise ValueError('Statistics need finite nonempty observations')
52    width = .25 if unit == '%' else .1
53    bins = Counter(math.floor(value / width) for value in values)
54    # Deterministic tie-break: the lowest interval among equally common bins.
55    modal = min(bins, key=lambda key: (-bins[key], key))
56    return dict(metric=field, unit=unit, count=len(values),
57                mean=statistics.mean(values), median=statistics.median(values),
58                stddev=statistics.stdev(values) if len(values) > 1 else 0.,
59                minimum=min(values), maximum=max(values),
60                **{'p' + str(p): percentile(values, p) for p in [1, 5, 10, 25, 50, 75, 90, 95, 99]},
61                modal_bin_low=modal * width, modal_bin_high=(modal + 1) * width,
62                modal_bin_count=bins[modal], modal_bin_width=width)
