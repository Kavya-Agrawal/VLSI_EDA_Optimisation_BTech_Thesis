35def replay(before, patch):
36    """Strict unified-diff replay used to validate every stored candidate."""
37    original = before.splitlines(keepends=True)
38    output = []
39    position = 0
40    body = patch.splitlines(keepends=True)
41    index = 2
42    while index < len(body):
43        match = re.match(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@', body[index])
44        if not match:
45            raise ValueError('Invalid patch hunk header')
46        start = max(0, int(match[1]) - 1)
47        if start < position:
48            raise ValueError('Overlapping patch hunks')
49        output.extend(original[position:start])
50        position = start
51        index += 1
52        while index < len(body) and not body[index].startswith('@@'):
53            line = body[index]
54            if line[0] in ' -':
55                if position >= len(original) or original[position] != line[1:]:
56                    raise ValueError('Patch context mismatch')
57                position += 1
58            if line[0] in ' +':
59                output.append(line[1:])
60            if line[0] not in ' +-':
61                raise ValueError('Unexpected patch line')
62            index += 1
63    output.extend(original[position:])
64    return ''.join(output)

67def config_variant(base, iteration, rng):
68    """Mix small edits with larger parameter-and-control-choice edits."""
69    fields = {
70        'load_weight': f'{rng.uniform(.6, 1.6):.8f}',
71        'density_weight': f'{rng.uniform(0, .6):.8f}',
72        'prefix_weight': f'{rng.uniform(0, .5):.8f}',
73        'neighborhood_weight': f'{rng.uniform(0, .4):.8f}',
74        'late_weight': f'{rng.uniform(0, .3):.8f}',
75        'critical_weight': f'{rng.uniform(0, .5):.8f}',
76        'budget': str(rng.randint(4, 40)),
77        'neighborhood': str(rng.randint(1, 16)),
78        'spatial_mode': str(rng.randrange(3)),
79        'use_dp': 'true' if rng.randrange(2) else 'false',
80    }
81    # Every seventh example is a small one-coefficient change.
82    chosen = ['load_weight'] if iteration % 7 == 0 else list(fields)
83    output = base
84    for field in chosen:
85        output, replacements = re.subn(
86            rf'(\b{field}\s*=\s*)[^;]+;', rf'\g<1>{fields[field]};', output)
87        if replacements != 1:
88            raise ValueError('Missing or ambiguous policy field: ' + field)
89    return output, chosen

92def simulated_labels(rng, iteration):
93    # No source-code effectiveness can be inferred from these invented labels.
94    # Exactly 70% near-zero, 22% moderate and 8% rare higher-impact labels.
95    bucket = (iteration - 1) % 100
96    low, high = (0., 1.) if bucket < 70 else ((1.5, 2.5) if bucket < 92 else (4., 5.))
97    wns = round(rng.uniform(low, high), 4)
98    tns = round(rng.uniform(low, high), 4)
99    runtime = round(rng.uniform(-.8, .3), 4)
100    short = round(.5 * wns + .5 * tns, 4)
101    long = round(short * rng.uniform(.5, 1.), 4)
102    score = round(.45 * wns + .45 * tns - .1 * runtime, 4)
103    return dict(simulated_wns_impact_pct=wns, simulated_tns_impact_pct=tns,
104                simulated_runtime_delta_s=runtime,
105                simulated_effectiveness=short,
106                simulated_short_term_impact=short,
107                simulated_long_term_impact=long,
108                simulated_overall_score=score,
109                simulated_accepted=wns > .2 and tns > .2 and score > .5,
110                agent_iteration=iteration, generation=(iteration - 1) // 50,
111                measured_openroad=False, applied_to_worktree=False,
112                label_provenance='fictional study labels; no OpenROAD evaluation')
